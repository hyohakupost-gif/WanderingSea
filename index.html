async function search() { const lang = localStorage.getItem('lang') || 'ja'; const name = document.getElementById('name').value.trim(); const title = document.getElementById('title').value.trim(); const resDiv = document.getElementById('results'); // 【重要】ここで既存の表示を完全にリセットする resDiv.innerHTML = ""; // 総当たり対策 if (name.length < 2 && title.length < 2) { resDiv.innerHTML = `

${translations[lang].error_input}

`; return; } // 検索中であることを示すメッセージを表示 resDiv.innerHTML = `

${translations[lang].status_searching}

`; try { const response = await fetch(`${SCRIPT_URL}?name=${encodeURIComponent(name)}&title=${encodeURIComponent(title)}`); const result = await response.json(); // 【重要】ここでもう一度クリアする（検索中メッセージを消すため） resDiv.innerHTML = ""; if (result.status === "error") { resDiv.innerHTML = `

${escapeHtml(result.message)}

`; } else if (result.data.length === 0) { resDiv.innerHTML = `

該当するお手紙が見つかりませんでした。

`; } else { let html = ""; result.data.forEach(item => { html += `
${escapeHtml(item.id)} content_copy
${translations[lang].label_date}${escapeHtml(item.date)}
${translations[lang].label_sender}${escapeHtml(item.name)}
${translations[lang].label_title}${escapeHtml(item.title)}
`; }); resDiv.innerHTML = html; } } catch (err) { resDiv.innerHTML = `

${translations[lang].error_connection}

`; } }
